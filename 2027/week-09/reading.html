final class ServiceRequest {

Concept and reasoning 11 / 19

Insertion and save form one user action

Swift excerpt · see practice download for context

let item = ServiceRequest(title: draft)
context.insert(item)
do {
    try context.save()
    draft = ""
} catch {
    context.rollback()
    message = "Could not save. Try again."
}
